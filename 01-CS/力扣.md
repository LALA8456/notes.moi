class Solution:
    def corpFlightBookings(self, bookings: list[list[int]], n: int) -> list[int]:
        #差分数组，对连续的数组进行加和减，把之前MN的复杂度变成M+N。公交车理论，左加右减
        #注意是从1-N的航班编号，类似前缀和也是创建多一个的数组，但是返回的是长为N的正常answer数组
        #这个和很多题 一样得画图，前面是定义公交，后面是前缀和的部分。上一次5，这次加了1，变成6
        diff=[0]*(n+1)  #因为last那里可能索引溢出 ，所以是N+1
        #bookings 二维，【【1，3，4】】从1到3号加4个座位
        for first,last,seats in bookings: # define those v in a for-loop
            diff[first-1]+=seats #first begin with 1
            diff[last]-=seats
        
        lastseats=0
        answer=[0]*n # 
        for i in range(n):
            answer[i]=lastseats+diff[i]
            lastseats=answer[i]
        
        return answer

class Solution:
    def twoSum(self, nums: list[int], target: int) -> list[int]:#target是变量名nums是变量 LIST这个数组
        #twosum是名字
        for i,item in enumerate(nums):#for最后是：   nums为数组没有括号
             for j in range(i+1,len(nums)):#range是下标的上下。enumerate是内容
                    ite=nums[j]#定义下标和内容
                    if item+ite==target:
                        return [i,j]#与item无关，返回下标！！

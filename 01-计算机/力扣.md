class Solution:
    def corpFlightBookings(self, bookings: list[list[int]], n: int) -> list[int]:
        #差分数组，对连续的数组进行加和减，把之前MN的复杂度变成M+N。公交车理论，左加右减
        #注意是从1-N的航班编号，类似前缀和也是创建多一个的数组，但是返回的是长为N的正常answer数组
        #这个和很多题 一样得画图，前面是定义公交，后面是前缀和的部分。上一次5，这次加了1，变成6
        diff=[0]*(n+1)  #因为last那里可能索引溢出 ，所以是N+1
        #bookings 二维，【【1，3，4】】从1到3号加4个座位
        for first,last,seats in bookings: # define those v in a for-loop
            diff[first-1]+=seats #first begin with 1
            diff[last]-=seats
        
        lastseats=0
        answer=[0]*n # 
        for i in range(n):
            answer[i]=lastseats+diff[i]
            lastseats=answer[i]
        
        return answer

class Solution:
    def dailyTemperatures(self, temperatures: list[int]) -> list[int]:

        #单调栈就是把之后用的到的信息存着
        #而且小的值会弹出，比较后大的到栈底。留存
        #代码：写创建一个stack，拿原数组长度新建一个answer数组
        #I遍历N，让当前温度等于遍历到的。  while中比较当前和历史，栈顶是-1，对应的temperature
        #如果大于栈顶（历史的），小的弹出。  对应下标为弹出的值，因为stack里面是下标。
        #遍历的I-这个弹出的下标，就是它们的间隔 等于 answer【弹出的下标】
        #while栈非空
        #else是当前的更小，按理说是pass。不是，是break，指针后移是break。
        #把当前的下标加入栈。因为我们栈初始化是0，所以不用写填0的。 空列表==【】，空指针==NOne数组
        #if stack  就是if stack非空，为1，会优化速度
        
        #单调栈，更高更小。最终自然有单调排序，不记谁高谁底，可以写暴力。
        
        stack=[]
        n=len(temperatures)
        answer=[0]*n
        for i in range(n):
            cur_temval=temperatures[i]
            while True:
                if stack and cur_temval>temperatures[(stack[-1])]:
                    pre_idx=stack.pop() #the index we want,so we can sub
                    answer[pre_idx]=i-pre_idx #answer 装index之差,aber its aws[pre idx]
                else:
                    break
            stack.append(i)#  I!
            
        return answer
